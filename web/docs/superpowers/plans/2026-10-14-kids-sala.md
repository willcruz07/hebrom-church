```ts
/** Conteúdo do QR do responsável: `HKIDS1:{uid}:{token}`. */
export const QR_PREFIX = 'HKIDS1'
/** 18 bytes aleatórios = 24 caracteres base64url. */
export const TOKEN_BYTES = 18

const BASE64URL = 'ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789-_'
const PART = /^[A-Za-z0-9_-]+$/

/** Token a partir de bytes aleatórios (quem chama usa `crypto.getRandomValues`). */
export function newToken(bytes: Uint8Array): string {
  let token = ''
  for (let i = 0; i + 2 < bytes.length; i += 3) {
    const n = (bytes[i] << 16) | (bytes[i + 1] << 8) | bytes[i + 2]
    token += BASE64URL[(n >> 18) & 63] + BASE64URL[(n >> 12) & 63] + BASE64URL[(n >> 6) & 63] + BASE64URL[n & 63]
  }
  return token
}

export const buildQrPayload = (uid: string, token: string) => `${QR_PREFIX}:${uid}:${token}`

export function parseQrPayload(text: string): { uid: string; token: string } | null {
  const parts = text.trim().split(':')
  if (parts.length !== 3 || parts[0] !== QR_PREFIX) return null
  const [, uid, token] = parts
  if (!PART.test(uid) || !PART.test(token)) return null
  return { uid, token }
}
```

```ts
import { OUTSIDE_ROOMS_LABEL, ROOMS, type RoomId } from './rooms.ts'

/** O que as regras precisam de um Timestamp do Firestore (permite testar sem Firebase). */
export interface TimestampLike {
  toMillis(): number
}

export type PassRoom = RoomId | 'outside'
export type PassStatus = 'in' | 'out' | 'no_checkout'
export type EntryVia = 'qr' | 'manual'

export interface Person {
  uid: string
  name: string
}

/** `uid` ausente = "Outra pessoa" (nome digitado pela tia). */
export interface PickedBy {
  uid?: string
  name: string
}

export interface StaffMember {
  uid: string
  name: string
  avatar_url: string | null
}

export interface PreviousPassOf<T> {
  room: PassRoom
  dropped_by: Person
  checked_in_at: T
  checked_out_at: T | null
  picked_by: PickedBy | null
}

/** `kids_sessions/{id}/attendance/{childId}` — uma passagem ativa por criança por sessão. */
export interface KidPassOf<T> {
  child_id: string
  child_name: string
  child_photo_url?: string
  birth_date: string
  room: PassRoom
  guardian_uids: string[]
  is_visitor: boolean
  status: PassStatus
  dropped_by: Person
  checked_in_at: T
  checked_in_by: Person
  checkin_via: EntryVia
  checked_out_at?: T
  checked_out_by?: Person
  checkout_via?: EntryVia
  picked_by?: PickedBy
  previous_passes?: PreviousPassOf<T>[]
}

/** `kids_sessions/{id}` — status público da Área Kids. */
export interface KidSessionOf<T> {
  id: string
  status: 'open' | 'closed'
  opened_at: T
  opened_by: Person
  closed_at?: T
  closed_by?: Person
  staff: StaffMember[]
  staff_uids: string[]
}

/** Aberta há mais que isso, a sessão aparece como fechada para os membros. */
export const SESSION_STALE_MS = 12 * 60 * 60 * 1000

export const ROOM_LABELS = {
  ...Object.fromEntries(ROOMS.map((room) => [room.id, room.label])),
  outside: OUTSIDE_ROOMS_LABEL,
} as Record<PassRoom, string>

export const isSessionStale = (openedAtMs: number, nowMs: number) => nowMs - openedAtMs > SESSION_STALE_MS

export function roomCounts(passes: { status: PassStatus; room: PassRoom }[]): Record<PassRoom, number> {
  const counts: Record<PassRoom, number> = { bercario: 0, kids: 0, juniores: 0, outside: 0 }
  for (const pass of passes) if (pass.status === 'in') counts[pass.room] += 1
  return counts
}

/** Texto de status da criança para quem está vendo (pai ou equipe). */
export function passLabel(
  pass: KidPassOf<TimestampLike> | null,
  viewerUid: string,
  formatTime: (ms: number) => string,
): string {
  if (!pass) return 'Fora da sala'
  const who = (person: { uid?: string; name: string }) => (person.uid && person.uid === viewerUid ? 'você' : person.name)
  if (pass.status === 'in') {
    return `Na sala ${ROOM_LABELS[pass.room]} desde ${formatTime(pass.checked_in_at.toMillis())} · deixado por ${who(pass.dropped_by)}`
  }
  if (pass.status === 'out') {
    const at = pass.checked_out_at ? ` às ${formatTime(pass.checked_out_at.toMillis())}` : ''
    return pass.picked_by ? `Retirado${at} por ${who(pass.picked_by)}` : `Retirado${at}`
  }
  return 'Saída não registrada'
}

/** Passagens que viram "saída não registrada" ao encerrar. */
export const passesToClose = <P extends { status: PassStatus }>(passes: P[]) => passes.filter((p) => p.status === 'in')
```

```ts
export function openSession(member: StaffMember): Promise<string> {
  return runTransaction(db, async (tx) => {
    const currentId = (await tx.get(metaDoc())).data()?.session_id
    if (typeof currentId === 'string' && currentId) {
      const current = await tx.get(sessionDoc(currentId))
      const data = current.data()
      if (data?.status === 'open') {
        const staff: StaffMember[] = Array.isArray(data.staff) ? data.staff : []
        if (!staff.some((s) => s.uid === member.uid)) {
          const next = [...staff, member]
          tx.update(current.ref, { staff: next, staff_uids: next.map((s) => s.uid) })
        }
        return currentId
      }
    }
    const ref = doc(collection(db, SESSIONS))
    tx.set(ref, {
      status: 'open',
      opened_at: Timestamp.now(),
      opened_by: { uid: member.uid, name: member.name },
      staff: [member],
      staff_uids: [member.uid],
    })
    tx.set(metaDoc(), { session_id: ref.id })
    return ref.id
  })
}

export async function setServing(sessionId: string, member: StaffMember, serving: boolean): Promise<void> {
  await runTransaction(db, async (tx) => {
    const snapshot = await tx.get(sessionDoc(sessionId))
    const data = snapshot.data()
    if (data?.status !== 'open') throw new SessionClosed()
    const others = (Array.isArray(data.staff) ? (data.staff as StaffMember[]) : []).filter((s) => s.uid !== member.uid)
    const next = serving ? [...others, member] : others
    tx.update(snapshot.ref, { staff: next, staff_uids: next.map((s) => s.uid) })
  })
}

  if (!remaining.empty) {

      closed_at: Timestamp.now(),
      closed_by: { uid: member.uid, name: member.name },

}

export async function checkIn(
  sessionId: string,
  entries: { child: KidChild; room: PassRoom }[],
  droppedBy: Person,
  staff: Person,
  via: EntryVia,
): Promise<void> {
  const now = Timestamp.now()
  await runTransaction(db, async (tx) => {
    if ((await tx.get(sessionDoc(sessionId))).data()?.status !== 'open') throw new SessionClosed()
    const refs = entries.map((e) => passDoc(sessionId, e.child.id))
    const existing = (await Promise.all(refs.map((ref) => tx.get(ref)))).map((s) => s.data() as KidPass | undefined)
    for (const pass of existing) {
      if (pass?.status === 'in') {
        throw new PassAlreadyIn(pass.child_name, pass.checked_in_by.name, pass.checked_in_at.toMillis())
      }
    }
    entries.forEach((entry, i) => {
      const input = { child: entry.child, room: entry.room, droppedBy, staff, via, now }
      const previous = existing[i]
      tx.set(refs[i], previous ? buildReentry(previous, input) : buildCheckIn(input))
    })
  })
}

export async function checkOut(
  sessionId: string,
  childIds: string[],
  pickedBy: PickedBy,
  staff: Person,
  via: EntryVia,
): Promise<void> {

  await runTransaction(db, async (tx) => {
    if ((await tx.get(sessionDoc(sessionId))).data()?.status !== 'open') throw new SessionClosed()
    const refs = childIds.map((id) => passDoc(sessionId, id))
    const snapshots = await Promise.all(refs.map((ref) => tx.get(ref)))
    snapshots.forEach((snapshot, i) => {
      if (snapshot.data()?.status === 'in') tx.update(refs[i], { ...buildCheckOut({ pickedBy, staff, via, now }) })
    })
  })
}
```

```ts
'use client'

import { useEffect, useState } from 'react'

/** Hora atual em ms, atualizada a cada `intervalMs`. Vale 0 até montar (evita `Date.now()` no render). */
export function useNow(intervalMs = 60_000): number {
  const [now, setNow] = useState(0)

  useEffect(() => {
    const tick = () => setNow(Date.now())
    const first = setTimeout(tick, 0)
    const timer = setInterval(tick, intervalMs)
    return () => {
      clearTimeout(first)
      clearInterval(timer)
    }
  }, [intervalMs])

  return now
}
```

```ts
  const key = [...new Set(uids)].sort().join(',')
  const [, setVersion] = useState(0)

  useEffect(() => {
    const missing = key ? key.split(',').filter((uid) => !cache.has(uid)) : []
    if (!missing.length) return
    let cancelled = false
    getGuardians(missing).then((found) => {
      for (const [uid, info] of Object.entries(found)) cache.set(uid, info)
      if (!cancelled) setVersion((v) => v + 1)
    })
    return () => {
      cancelled = true
    }
  }, [key])

  return Object.fromEntries(uids.map((uid) => [uid, cache.get(uid)]))
}
```

```tsx
import { useEffect, useState } from 'react'

export function MyQrCard() {
  const { currentUser } = useAuth()
  const { confirm } = useMessages()
  const uid = currentUser?.uid
  const [token, setToken] = useState<{ uid: string; value: string } | null>(null)
  const [failed, setFailed] = useState(false)
  const [renewing, setRenewing] = useState(false)

  useEffect(() => {
    if (!uid) return
    let cancelled = false
    getOrCreateQrToken(uid)
      .then((value) => {
        if (!cancelled) setToken({ uid, value })
      })
      .catch((error) => {
        console.error(error)
        if (!cancelled) setFailed(true)
      })
    return () => {
      cancelled = true
    }
  }, [uid])

  const renew = async () => {
    if (!uid) return
    const confirmed = await confirm({
      title: 'Gerar novo QR',
      description: 'O QR atual deixará de funcionar.',
      confirmText: 'Gerar novo',
    })
    if (!confirmed) return
    setRenewing(true)
    try {
      setToken({ uid, value: await regenerateQrToken(uid) })
      toast.success('Novo QR gerado')
    } catch (error) {
      console.error(error)
      toast.error('Sem conexão. Tente de novo.')
    } finally {
      setRenewing(false)
    }
  }

  const current = token && token.uid === uid ? token.value : null

  return (
    <section className="space-y-4 rounded-2xl border border-slate-200 bg-white p-6 text-center shadow-sm dark:border-slate-800 dark:bg-slate-900/50">
```

```ts
import { FirebaseError } from 'firebase/app'
import dayjs from '@/lib/dayjs'
import { PassAlreadyIn, SessionClosed } from '@/services/firebase/kids-room'

/** Mensagem do toast para falhas da Sala. */
export function roomErrorMessage(error: unknown): string {
  if (error instanceof PassAlreadyIn) {
    return `${error.childName} já foi registrado por ${error.byName} às ${dayjs(error.atMs).format('HH:mm')}.`
  }
  if (error instanceof SessionClosed) return 'A Área Kids foi encerrada.'
  if (error instanceof FirebaseError && error.code === 'permission-denied') {
    return 'A Área Kids foi encerrada ou você não está servindo.'
  }
  return 'Sem conexão. Tente de novo.'
}
```

```tsx
'use client'

import { useState } from 'react'
import { AlertTriangle } from 'lucide-react'
import { Button } from '@/components/ui/button'
import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from '@/components/ui/dialog'
import { HebromSpinner } from '@/components/ui/HebromSpinner'
import { Input } from '@/components/ui/input'
import type { PickedBy } from '@/lib/kids/session'
import { cn } from '@/lib/utils'
import type { KidPass } from '@/types'
import { useGuardianInfos } from '@/components/kids/useGuardianInfos'

const OTHER = '__other__'

interface ManualCheckOutDialogProps {
  pass: KidPass | null
  open: boolean
  onOpenChange: (open: boolean) => void
  onConfirm: (picked: PickedBy) => Promise<boolean>
}

/** Saída sem QR: um dos responsáveis ou "Outra pessoa" (nome digitado). */
export function ManualCheckOutDialog({ pass, open, onOpenChange, onConfirm }: ManualCheckOutDialogProps) {
  const guardians = useGuardianInfos(pass?.guardian_uids ?? [])
  const [choice, setChoice] = useState('')
  const [otherName, setOtherName] = useState('')
  const [saving, setSaving] = useState(false)

  const [wasOpen, setWasOpen] = useState(open)
  if (open !== wasOpen) {
    setWasOpen(open)
    if (open) {
      setChoice(pass?.guardian_uids[0] ?? OTHER)
      setOtherName('')
    }
  }

  const options = (pass?.guardian_uids ?? []).map((uid) => ({ uid, name: guardians[uid]?.name ?? 'Responsável' }))
  const canConfirm = choice === OTHER ? otherName.trim().length > 0 : !!choice

  const confirm = async () => {
    const picked: PickedBy =
      choice === OTHER ? { name: otherName.trim() } : { uid: choice, name: options.find((o) => o.uid === choice)?.name ?? 'Responsável' }
    setSaving(true)
    const ok = await onConfirm(picked)
    setSaving(false)
    if (ok) onOpenChange(false)
  }

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent variant="sheet" className="sm:max-w-[460px]">
        <DialogHeader>
          <DialogTitle className="text-base font-black md:text-xl">Registrar saída</DialogTitle>
          <DialogDescription>{pass ? `Quem está buscando ${pass.child_name}?` : ''}</DialogDescription>
        </DialogHeader>

        <div className="space-y-2">
          {options.map((o) => (
            <Choice key={o.uid} active={choice === o.uid} onClick={() => setChoice(o.uid)}>
              {o.name}
            </Choice>
          ))}
          <Choice active={choice === OTHER} onClick={() => setChoice(OTHER)}>
            Outra pessoa
          </Choice>
          {choice === OTHER && (
            <div className="space-y-2">
              <Input
                autoFocus
                value={otherName}
                onChange={(e) => setOtherName(e.target.value)}
                maxLength={80}
                placeholder="Nome de quem está buscando"
                className="rounded-xl"
              />
              <p className="flex items-center gap-1.5 text-xs font-semibold text-amber-700 dark:text-amber-400">
                <AlertTriangle className="h-3.5 w-3.5" /> Confirme a identidade de quem está buscando.
              </p>
            </div>
          )}
        </div>

        <DialogFooter>
          <Button
            type="button"
            onClick={confirm}
            disabled={!canConfirm || saving}
            className="h-11 w-full rounded-xl bg-amber-600 px-8 font-black text-white shadow-lg shadow-amber-500/25 hover:bg-amber-700 active:scale-95 sm:w-auto"
          >
            {saving ? <HebromSpinner size="sm" className="brightness-200" /> : 'Registrar saída'}
          </Button>
          <Button
            type="button"
            variant="ghost"
            onClick={() => onOpenChange(false)}
            className="h-11 w-full rounded-xl font-bold text-slate-500 hover:bg-slate-100 dark:hover:bg-slate-800 sm:w-auto"
          >
            Cancelar
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  )
}

function Choice({ active, onClick, children }: { active: boolean; onClick: () => void; children: React.ReactNode }) {
  return (
    <button
      type="button"
      onClick={onClick}
      aria-pressed={active}
      className={cn(
        'flex h-12 w-full items-center rounded-xl border px-4 text-left text-sm font-bold transition-colors',
        active
          ? 'border-amber-600 bg-amber-50 text-amber-800 dark:bg-amber-900/20 dark:text-amber-300'
          : 'border-slate-200 text-slate-700 hover:border-amber-300 dark:border-slate-800 dark:text-slate-200',
      )}
    >
      {children}
    </button>
  )
}
```

```tsx
'use client'

import { useEffect, useMemo, useState } from 'react'
import { ChevronDown, LogIn, LogOut, QrCode } from 'lucide-react'
import { toast } from 'sonner'
import { KidDetailModal } from '@/components/kids/KidDetailModal'
import { useGuardianInfos } from '@/components/kids/useGuardianInfos'
import { Button } from '@/components/ui/button'
import { HebromSpinner } from '@/components/ui/HebromSpinner'
import { useMessages } from '@/hooks/useMessages'
import dayjs from '@/lib/dayjs'
import { ROOMS, type RoomId } from '@/lib/kids/rooms'
import { passesToClose, ROOM_LABELS, roomCounts, type EntryVia, type PassRoom, type Person, type PickedBy } from '@/lib/kids/session'
import { cn } from '@/lib/utils'
import {
  changeRoom,
  checkIn,
  checkOut,
  closeSession,
  openSession,
  setServing,
  subscribeToSessionPasses,
} from '@/services/firebase/kids-room'
import { useAuth } from '@/store/useAuth'
import { useKidsStore } from '@/store/useKidsStore'
import type { KidChild, KidPass, KidSession } from '@/types'
import { CheckInSheet } from './CheckInSheet'
import { CheckOutSheet } from './CheckOutSheet'
import { ManualCheckOutDialog } from './ManualCheckOutDialog'
import { PassCard } from './PassCard'
import { RoomChips } from './RoomChips'
import { roomErrorMessage } from './room-errors'
import { SignupQrDialog } from './SignupQrDialog'

type Filter = 'all' | RoomId

interface RoomPanelProps {
  session: KidSession | null
  /** Aberta há mais de 12 h: pede para encerrar. */
  stale: boolean
}

/** Sala da equipe Kids: abrir/encerrar, servir, entrada, saída e lista. */
export function RoomPanel({ session, stale }: RoomPanelProps) {
  const { currentUser } = useAuth()
  const { confirm } = useMessages()
  const children = useKidsStore((s) => s.children)
  const startKids = useKidsStore((s) => s.startListening)
  const [passState, setPassState] = useState<{ sessionId: string | null; passes: KidPass[] }>({ sessionId: null, passes: [] })
  const [filter, setFilter] = useState<Filter>('all')
  const [busy, setBusy] = useState(false)
  const [sheet, setSheet] = useState<'in' | 'out' | null>(null)
  const [signupOpen, setSignupOpen] = useState(false)
  const [showOut, setShowOut] = useState(false)
  // ids ficam guardados ao fechar, para o conteúdo não sumir durante a animação de saída
  const [detail, setDetail] = useState<{ childId: string | null; open: boolean }>({ childId: null, open: false })
  const [manual, setManual] = useState<{ childId: string | null; open: boolean }>({ childId: null, open: false })

  useEffect(() => {
    startKids()
  }, [startKids])

  const sessionId = session?.id ?? null
  useEffect(() => {
    if (!sessionId) return
    return subscribeToSessionPasses(
      sessionId,
      (passes) => setPassState({ sessionId, passes }),
      (error) => {
        console.error(error)
        toast.error('Erro ao carregar a sala')
      },
    )
  }, [sessionId])

  const passes = passState.sessionId === sessionId ? passState.passes : []
  const childById = useMemo(() => new Map(children.map((c) => [c.id, c])), [children])
  const counts = roomCounts(passes)
  const inRoom = passes
    .filter((p) => p.status === 'in' && (filter === 'all' || p.room === filter))
    .sort((a, b) => a.child_name.localeCompare(b.child_name, 'pt-BR'))
  const left = passes
    .filter((p) => p.status !== 'in')
    .sort((a, b) => (b.checked_out_at?.toMillis() ?? 0) - (a.checked_out_at?.toMillis() ?? 0))

  const detailPass = passes.find((p) => p.child_id === detail.childId) ?? null
  const detailChild = detail.childId ? childById.get(detail.childId) ?? null : null
  const detailGuardians = useGuardianInfos(detailChild?.guardian_uids ?? [])
  const manualPass = passes.find((p) => p.child_id === manual.childId) ?? null

  if (!currentUser) return null
  const me: Person = { uid: currentUser.uid, name: currentUser.profile.full_name }
  const member = { ...me, avatar_url: currentUser.profile.avatar_url }
  const serving = !!session && session.staff_uids.includes(me.uid)

  const run = async (action: () => Promise<unknown>, success: string) => {
    setBusy(true)
    try {
      await action()
      toast.success(success)
      return true
    } catch (error) {
      console.error(error)
      toast.error(roomErrorMessage(error))
      return false
    } finally {
      setBusy(false)
    }
  }

  const close = async () => {
    if (!session) return
    const remaining = passesToClose(passes).length
    const confirmed = await confirm({
      title: 'Encerrar Área Kids',
      description: remaining
        ? `${remaining} crianças ainda estão na sala e ficarão como saída não registrada.`
        : 'Encerrar a Área Kids?',
      confirmText: 'Encerrar',
      destructive: true,
    })

  }

  const registerEntries = (entries: { child: KidChild; room: PassRoom }[], droppedBy: Person, via: EntryVia) =>
    session
      ? run(
          () => checkIn(session.id, entries, droppedBy, me, via),
          entries.length === 1 ? 'Entrada registrada' : `${entries.length} entradas registradas`,
        )
      : Promise.resolve(false)

  const registerExit = (childIds: string[], picked: PickedBy, via: EntryVia) =>
    session
      ? run(
          () => checkOut(session.id, childIds, picked, me, via),
          childIds.length === 1 ? 'Saída registrada' : `${childIds.length} saídas registradas`,
        )
      : Promise.resolve(false)

  if (!session) {
    return (
      <div className="space-y-4 rounded-2xl border-2 border-dashed border-amber-200 p-8 text-center dark:border-amber-900/50">
        <p className="text-sm text-slate-500">A Área Kids está fechada.</p>

        >
          {busy ? <HebromSpinner size="sm" className="brightness-200" /> : 'Abrir Área Kids'}
        </Button>
      </div>
    )
  }

  if (stale) {
    return (
      <div className="space-y-4 rounded-2xl border border-amber-200 bg-amber-50/60 p-6 dark:border-amber-900/50 dark:bg-amber-900/10">
        <p className="font-semibold text-amber-900 dark:text-amber-300">
          A Área Kids está aberta desde {dayjs(session.opened_at.toDate()).format('ddd DD/MM [às] HH:mm')}. Encerrar?
        </p>
        <Button
          type="button"
          disabled={busy}
          onClick={close}
          className="h-11 rounded-xl bg-amber-600 px-8 font-black text-white hover:bg-amber-700"
        >
          Encerrar
        </Button>
      </div>
    )
  }

  const tile = (label: string, value: number, active: boolean, onClick: () => void) => (
    <button
      type="button"
      onClick={onClick}
      className={cn(
        'rounded-xl px-3 py-2 text-left transition-all active:scale-95',
        active ? 'bg-amber-600 text-white shadow-md shadow-amber-500/25' : 'bg-white text-slate-600 ring-1 ring-slate-200 dark:bg-slate-900 dark:text-slate-300 dark:ring-slate-800',
      )}
    >
      <span className="block text-[10px] font-bold uppercase tracking-wider opacity-80">{label}</span>
      <span className="block text-lg font-black">{value}</span>
    </button>
  )

  return (
    <div className="space-y-5">
      <div className="flex flex-wrap items-center gap-2">
        <Button
          type="button"
          variant="ghost"
          disabled={busy}
          onClick={() => run(() => setServing(session.id, member, !serving), serving ? 'Você saiu do serviço' : 'Você está servindo')}
          className={cn(
            'h-11 rounded-xl px-5 font-bold',
            serving
              ? 'bg-emerald-50 text-emerald-700 hover:bg-emerald-100 dark:bg-emerald-900/20 dark:text-emerald-400'
              : 'bg-amber-50 text-amber-800 hover:bg-amber-100 dark:bg-amber-900/20 dark:text-amber-300',
          )}
        >
          {serving ? 'Sair do serviço' : 'Estou servindo'}
        </Button>
        <Button
          type="button"
          variant="ghost"
          onClick={() => setSignupOpen(true)}
          className="h-11 gap-2 rounded-xl font-bold text-slate-500 hover:bg-slate-100 dark:hover:bg-slate-800"
        >
          <QrCode className="h-4 w-4" /> QR para criar conta
        </Button>
        <Button
          type="button"
          variant="ghost"
          disabled={busy}
          onClick={close}
          className="ml-auto h-11 rounded-xl font-bold text-red-600 hover:bg-red-50 hover:text-red-600 dark:text-red-400 dark:hover:bg-red-900/20"
        >
          Encerrar
        </Button>
      </div>

      {!serving && (
        <p className="rounded-xl bg-amber-50 p-3 text-sm font-medium text-amber-800 dark:bg-amber-900/15 dark:text-amber-300">
          Toque em Estou servindo para registrar entradas e saídas.

      <div className="grid grid-cols-2 gap-3">
        <Button
          type="button"
          disabled={!serving || busy}
          onClick={() => setSheet('in')}

        >
          <LogIn className="h-5 w-5" /> Entrada
        </Button>
        <Button
          type="button"
          disabled={!serving || busy}
          onClick={() => setSheet('out')}
          className="h-16 gap-2 rounded-2xl bg-slate-900 text-base font-black text-white shadow-lg hover:bg-slate-800 active:scale-95 dark:bg-slate-700 dark:hover:bg-slate-600"
        >
          <LogOut className="h-5 w-5" /> Saída
        </Button>
      </div>

      <div className="no-scrollbar -mx-4 flex gap-2 overflow-x-auto px-4 pb-1 sm:mx-0 sm:px-0">
        {tile('Todas', counts.bercario + counts.kids + counts.juniores + counts.outside, filter === 'all', () => setFilter('all'))}

          Nenhuma criança na sala{filter === 'all' ? '' : ` ${ROOM_LABELS[filter]}`}.

              pass={pass}
```

```tsx
'use client'

import { useEffect, useRef, useState, type ReactNode } from 'react'
import { CameraOff } from 'lucide-react'

interface QrScannerViewProps {
  /** Chamado uma vez por leitura; para ler de novo, o pai remonta o componente (`key`). */
  onResult: (text: string) => void
  /** Conteúdo abaixo da câmera (ex.: "Buscar pelo nome"). */
  footer?: ReactNode
}

/** Câmera traseira lendo QR (qr-scanner: funciona no Safari/iOS, inclusive PWA). */
export function QrScannerView({ onResult, footer }: QrScannerViewProps) {
  const videoRef = useRef<HTMLVideoElement>(null)
  const onResultRef = useRef(onResult)
  const [cameraError, setCameraError] = useState(false)

  useEffect(() => {
    onResultRef.current = onResult
  }, [onResult])

  useEffect(() => {
    const video = videoRef.current
    if (!video) return
    let cancelled = false
    let handled = false
    let scanner: { start(): Promise<void>; stop(): void; destroy(): void } | null = null

    import('qr-scanner')
      .then(({ default: QrScanner }) => {
        if (cancelled) return
        const instance = new QrScanner(
          video,
          (result) => {
            if (handled) return
            handled = true
            instance.stop()
            onResultRef.current(result.data)
          },
          { preferredCamera: 'environment', highlightScanRegion: true, highlightCodeOutline: true, returnDetailedScanResult: true },
        )
        scanner = instance
        return instance.start()
      })
      .catch((error) => {
        console.error(error)
        if (!cancelled) setCameraError(true)
      })

    return () => {
      cancelled = true
      scanner?.destroy()
    }
  }, [])

  return (
    <div className="space-y-3">
      {cameraError ? (
        <div className="space-y-2 rounded-2xl border-2 border-dashed border-slate-200 p-6 text-center dark:border-slate-800">
          <CameraOff className="mx-auto h-8 w-8 text-slate-400" />
          <p className="text-sm font-semibold text-slate-700 dark:text-slate-200">Não foi possível usar a câmera.</p>
          <p className="text-xs text-slate-500">
            Libere o acesso à câmera para este app nas configurações do navegador (no iPhone: Ajustes → Safari → Câmera) e
            tente de novo, ou busque a criança pelo nome.
          </p>
        </div>
      ) : (
        <div className="relative mx-auto aspect-square w-full max-w-sm overflow-hidden rounded-2xl bg-slate-900">
          <video ref={videoRef} playsInline muted className="h-full w-full object-cover" />
        </div>
      )}
      {footer}
    </div>
  )
}
```

```tsx
import { useState } from 'react'
```
```tsx
import { useGuardianInfos } from '@/components/kids/useGuardianInfos'
import { Button } from '@/components/ui/button'
import { Dialog, DialogContent, DialogDescription, DialogHeader, DialogTitle } from '@/components/ui/dialog'
import { HebromSpinner } from '@/components/ui/HebromSpinner'
```

```tsx
import type { PickedBy } from '@/lib/kids/session'
import { cn } from '@/lib/utils'
import { resolveQr } from '@/services/firebase/kids-room'
import type { KidPass } from '@/types'
import { QrScannerView } from './QrScannerView'

export interface CheckOutSheetProps {
  open: boolean
  onOpenChange: (open: boolean) => void
  passes: KidPass[]
  onConfirm: (childIds: string[], picked: PickedBy) => Promise<boolean>
}

/** Saída pelo QR de quem veio buscar. */
export function CheckOutSheet({ open, onOpenChange, passes, onConfirm }: CheckOutSheetProps) {
  const [adultUid, setAdultUid] = useState<string | null>(null)
  const [scanKey, setScanKey] = useState(0)
  const [checking, setChecking] = useState(false)
  const [message, setMessage] = useState('')
  const [selected, setSelected] = useState<string[]>([])
  const [saving, setSaving] = useState(false)
  const guardians = useGuardianInfos(adultUid ? [adultUid] : [])

  const [wasOpen, setWasOpen] = useState(open)
  if (open !== wasOpen) {
    setWasOpen(open)
    if (open) {
      setAdultUid(null)
      setMessage('')
      setSelected([])
    }
  }

  const inRoomOf = (uid: string) => passes.filter((p) => p.status === 'in' && p.guardian_uids.includes(uid))
  const mine = adultUid ? inRoomOf(adultUid) : []

  const onScan = async (text: string) => {
    setMessage('')
    setChecking(true)
    try {
      const result = await resolveQr(text)
      if (!result.ok) {
        setMessage(result.reason === 'invalid' ? 'QR não reconhecido.' : 'Este QR foi substituído. Peça para a pessoa abrir o QR atual no app.')
      } else if (inRoomOf(result.uid).length === 0) {
        setMessage('Nenhuma criança deste responsável está na sala.')
      } else {
        setAdultUid(result.uid)
        setSelected(inRoomOf(result.uid).map((p) => p.child_id))
        return
      }
    } catch (error) {
      console.error(error)
      setMessage('Sem conexão. Tente de novo.')
    } finally {
      setChecking(false)
    }
    setScanKey((k) => k + 1)
  }

  const confirm = async () => {
    if (!adultUid || !selected.length) return
    setSaving(true)
    const ok = await onConfirm(selected, { uid: adultUid, name: guardians[adultUid]?.name ?? 'Responsável' })
    setSaving(false)
    if (ok) onOpenChange(false)
  }

  const adult = adultUid ? guardians[adultUid] : null

  return (
    <Dialog open={open} onOpenChange={onOpenChange}>
      <DialogContent variant="sheet" className="sm:max-w-[520px]">
        <DialogHeader>
          <DialogTitle className="text-base font-black md:text-xl">Saída</DialogTitle>
          <DialogDescription>
            {adultUid ? 'Confirme as crianças que estão saindo.' : 'Leia o QR de quem veio buscar.'}
          </DialogDescription>
        </DialogHeader>

        {!adultUid ? (
          <QrScannerView
            key={scanKey}
            onResult={onScan}
            footer={
              <div className="space-y-2">
                {checking && (
                  <div className="flex justify-center">
                    <HebromSpinner size="sm" />
                  </div>
                )}
                {message && <p className="text-center text-sm font-semibold text-red-500">{message}</p>}
                <p className="text-center text-xs text-slate-500">
                  Sem QR? Toque na criança na lista da sala e use Registrar saída.
                </p>
              </div>
            }
          />
        ) : (
          <div className="space-y-4">
            <div className="flex items-center gap-3 rounded-xl bg-amber-50 p-3 dark:bg-amber-900/15">
              <UserAvatar src={adult?.avatar_url} name={adult?.name} size={40} />
              <span className="truncate text-sm font-bold text-slate-900 dark:text-white">{adult?.name ?? '…'}</span>
            </div>
            <ul className="space-y-2">
              {mine.map((pass) => {
                const checked = selected.includes(pass.child_id)
                return (
                  <li key={pass.child_id}>
                    <button
                      type="button"
                      onClick={() =>
                        setSelected((s) => (checked ? s.filter((id) => id !== pass.child_id) : [...s, pass.child_id]))
                      }
                      aria-pressed={checked}
                      className={cn(
                        'flex w-full items-center gap-3 rounded-xl border p-3 text-left',
                        checked ? 'border-amber-500 bg-amber-50/50 dark:bg-amber-900/10' : 'border-slate-200 dark:border-slate-800',
                      )}
                    >
                      <span
                        className={cn(
                          'flex h-6 w-6 shrink-0 items-center justify-center rounded-md border-2 text-xs font-black',
                          checked ? 'border-amber-600 bg-amber-600 text-white' : 'border-slate-300',
                        )}
                      >
                        {checked ? '✓' : ''}
                      </span>
                      <UserAvatar src={pass.child_photo_url} name={pass.child_name} size={36} />
                      <span className="truncate text-sm font-bold">{pass.child_name}</span>
                    </button>
                  </li>
                )
              })}
            </ul>
            <div className="flex flex-col gap-3 sm:flex-row sm:justify-end">
              <Button
                type="button"
                onClick={confirm}
                disabled={saving || selected.length === 0}
                className="h-11 w-full rounded-xl bg-amber-600 px-8 font-black text-white shadow-lg shadow-amber-500/25 hover:bg-amber-700 active:scale-95 sm:w-auto"
              >
                {saving ? <HebromSpinner size="sm" className="brightness-200" /> : `Registrar saída (${selected.length})`}
              </Button>
              <Button
                type="button"
                variant="ghost"
                onClick={() => {
                  setAdultUid(null)
                  setScanKey((k) => k + 1)
                }}
                className="h-11 w-full rounded-xl font-bold text-slate-500 hover:bg-slate-100 dark:hover:bg-slate-800 sm:w-auto"
              >
                Ler outro QR
              </Button>
            </div>
          </div>
        )}
      </DialogContent>
    </Dialog>
  )
}
```
